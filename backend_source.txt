from django.contrib.auth.models import AbstractUser
from django.db import models


class Student(AbstractUser):
    """Custom user model for students"""

    STUDY_PREFERENCE_CHOICES = [
        ('morning', 'Morning (6AM - 12PM)'),
        ('afternoon', 'Afternoon (12PM - 6PM)'),
        ('evening', 'Evening (6PM - 12AM)'),
        ('night', 'Night (12AM - 6AM)'),
        ('flexible', 'Flexible'),
    ]

    YEAR_CHOICES = [
        (1, 'First Year'),
        (2, 'Second Year'),
        (3, 'Third Year'),
        (4, 'Fourth Year'),
    ]

    student_number = models.CharField(max_length=20, unique=True, blank=True, null=True)
    program = models.CharField(max_length=100, blank=True)
    year_of_study = models.PositiveSmallIntegerField(choices=YEAR_CHOICES, default=1)
    phone = models.CharField(max_length=20, blank=True)

    # Study preferences for AI scheduling
    study_preference = models.CharField(
        max_length=20, 
        choices=STUDY_PREFERENCE_CHOICES, 
        default='flexible'
    )
    max_study_hours_per_day = models.PositiveSmallIntegerField(default=6)
    preferred_break_duration = models.PositiveSmallIntegerField(default=30)  # minutes

    # Notification settings
    email_notifications = models.BooleanField(default=True)
    push_notifications = models.BooleanField(default=True)

    created_at = models.DateTimeField(auto_now_add=True)
    updated_at = models.DateTimeField(auto_now=True)

    class Meta:
        db_table = 'students'
        ordering = ['-created_at']

    def __str__(self):
        return f"{self.first_name} {self.last_name} ({self.student_number or self.username})"

    def get_full_name(self):
        return f"{self.first_name} {self.last_name}".strip() or self.username

from rest_framework import serializers
from django.contrib.auth import get_user_model
from django.contrib.auth.password_validation import validate_password

Student = get_user_model()


class StudentSerializer(serializers.ModelSerializer):
    """Student profile serializer"""
    full_name = serializers.CharField(source='get_full_name', read_only=True)

    class Meta:
        model = Student
        fields = [
            'id', 'username', 'email', 'first_name', 'last_name', 'full_name',
            'student_number', 'program', 'year_of_study', 'phone',
            'study_preference', 'max_study_hours_per_day', 'preferred_break_duration',
            'email_notifications', 'push_notifications',
            'created_at', 'updated_at'
        ]
        read_only_fields = ['id', 'created_at', 'updated_at']


class StudentRegistrationSerializer(serializers.ModelSerializer):
    """Registration serializer with password validation"""
    password = serializers.CharField(write_only=True, required=True, validators=[validate_password])
    password_confirm = serializers.CharField(write_only=True, required=True)

    class Meta:
        model = Student
        fields = [
            'username', 'email', 'password', 'password_confirm',
            'first_name', 'last_name', 'student_number', 'program',
            'year_of_study', 'study_preference'
        ]

    def validate(self, attrs):
        if attrs['password'] != attrs['password_confirm']:
            raise serializers.ValidationError({"password": "Password fields didn't match."})
        return attrs

    def create(self, validated_data):
        validated_data.pop('password_confirm')
        student = Student.objects.create_user(**validated_data)
        return student


class StudentLoginSerializer(serializers.Serializer):
    """Login serializer"""
    username = serializers.CharField(required=True)
    password = serializers.CharField(required=True, write_only=True)


class PasswordChangeSerializer(serializers.Serializer):
    """Password change serializer"""
    old_password = serializers.CharField(required=True, write_only=True)
    new_password = serializers.CharField(required=True, write_only=True, validators=[validate_password])
    new_password_confirm = serializers.CharField(required=True, write_only=True)

    def validate(self, attrs):
        if attrs['new_password'] != attrs['new_password_confirm']:
            raise serializers.ValidationError({"new_password": "Password fields didn't match."})
        return attrs

from rest_framework import status, generics
from rest_framework.decorators import api_view, permission_classes
from rest_framework.permissions import AllowAny, IsAuthenticated
from rest_framework.response import Response
from rest_framework.authtoken.models import Token
from django.contrib.auth import authenticate, get_user_model
from django.contrib.auth.hashers import check_password

from .serializers import (
    StudentSerializer, 
    StudentRegistrationSerializer, 
    StudentLoginSerializer,
    PasswordChangeSerializer
)

Student = get_user_model()


class RegisterView(generics.CreateAPIView):
    """Student registration endpoint"""
    queryset = Student.objects.all()
    serializer_class = StudentRegistrationSerializer
    permission_classes = [AllowAny]

    def create(self, request, *args, **kwargs):
        serializer = self.get_serializer(data=request.data)
        serializer.is_valid(raise_exception=True)
        student = serializer.save()
        token, created = Token.objects.get_or_create(user=student)
        return Response({
            'user': StudentSerializer(student).data,
            'token': token.key,
            'message': 'Registration successful'
        }, status=status.HTTP_201_CREATED)


class LoginView(generics.GenericAPIView):
    """Student login endpoint"""
    serializer_class = StudentLoginSerializer
    permission_classes = [AllowAny]

    def post(self, request):
        serializer = self.get_serializer(data=request.data)
        serializer.is_valid(raise_exception=True)

        username = serializer.validated_data['username']
        password = serializer.validated_data['password']

        student = authenticate(username=username, password=password)

        if student:
            token, created = Token.objects.get_or_create(user=student)
            return Response({
                'user': StudentSerializer(student).data,
                'token': token.key,
                'message': 'Login successful'
            })

        return Response(
            {'error': 'Invalid credentials'}, 
            status=status.HTTP_401_UNAUTHORIZED
        )


class LogoutView(generics.GenericAPIView):
    """Student logout endpoint"""
    permission_classes = [IsAuthenticated]

    def post(self, request):
        request.user.auth_token.delete()
        return Response({'message': 'Logout successful'})


class ProfileView(generics.RetrieveUpdateAPIView):
    """Student profile endpoint"""
    serializer_class = StudentSerializer
    permission_classes = [IsAuthenticated]

    def get_object(self):
        return self.request.user


class PasswordChangeView(generics.GenericAPIView):
    """Password change endpoint"""
    serializer_class = PasswordChangeSerializer
    permission_classes = [IsAuthenticated]

    def post(self, request):
        serializer = self.get_serializer(data=request.data)
        serializer.is_valid(raise_exception=True)

        student = request.user
        if not check_password(serializer.validated_data['old_password'], student.password):
            return Response(
                {'error': 'Current password is incorrect'},
                status=status.HTTP_400_BAD_REQUEST
            )

        student.set_password(serializer.validated_data['new_password'])
        student.save()

        # Delete old token and create new one
        Token.objects.filter(user=student).delete()
        token = Token.objects.create(user=student)

        return Response({
            'message': 'Password changed successfully',
            'token': token.key
        })

from django.db import models
from django.conf import settings


class AttendanceRecord(models.Model):
    """Individual attendance record for a class session"""

    STATUS_CHOICES = [
        ('present', 'Present'),
        ('absent', 'Absent'),
        ('late', 'Late'),
        ('excused', 'Excused'),
        ('cancelled', 'Class Cancelled'),
    ]

    student = models.ForeignKey(settings.AUTH_USER_MODEL, on_delete=models.CASCADE, related_name='attendance_records')
    class_schedule = models.ForeignKey('timetable.ClassSchedule', on_delete=models.CASCADE, related_name='attendance_records')

    date = models.DateField()
    status = models.CharField(max_length=20, choices=STATUS_CHOICES, default='present')
    check_in_time = models.TimeField(null=True, blank=True)
    check_out_time = models.TimeField(null=True, blank=True)

    # For late arrivals
    minutes_late = models.PositiveSmallIntegerField(default=0)

    # Notes
    notes = models.TextField(blank=True)

    created_at = models.DateTimeField(auto_now_add=True)
    updated_at = models.DateTimeField(auto_now=True)

    class Meta:
        db_table = 'attendance_records'
        unique_together = ['student', 'class_schedule', 'date']
        ordering = ['-date']

    def __str__(self):
        return f"{self.student.username} - {self.class_schedule.course.code} ({self.date}: {self.status})"


class AttendanceSummary(models.Model):
    """Rolling attendance summary per course per student"""

    student = models.ForeignKey(settings.AUTH_USER_MODEL, on_delete=models.CASCADE, related_name='attendance_summaries')
    course = models.ForeignKey('timetable.Course', on_delete=models.CASCADE, related_name='attendance_summaries')

    total_sessions = models.PositiveIntegerField(default=0)
    present_count = models.PositiveIntegerField(default=0)
    absent_count = models.PositiveIntegerField(default=0)
    late_count = models.PositiveIntegerField(default=0)
    excused_count = models.PositiveIntegerField(default=0)

    attendance_percentage = models.DecimalField(max_digits=5, decimal_places=2, default=100.00)

    # Thresholds
    minimum_required_percentage = models.DecimalField(max_digits=5, decimal_places=2, default=80.00)
    warning_threshold_percentage = models.DecimalField(max_digits=5, decimal_places=2, default=85.00)

    # Alert tracking
    last_warning_sent = models.DateTimeField(null=True, blank=True)
    warning_count = models.PositiveSmallIntegerField(default=0)

    updated_at = models.DateTimeField(auto_now=True)

    class Meta:
        db_table = 'attendance_summaries'
        unique_together = ['student', 'course']

    def __str__(self):
        return f"{self.student.username} - {self.course.code}: {self.attendance_percentage}%"

    def calculate_percentage(self):
        """Recalculate attendance percentage"""
        countable = self.total_sessions - self.excused_count
        if countable == 0:
            self.attendance_percentage = 100.00
        else:
            self.attendance_percentage = round(
                (self.present_count + self.late_count) / countable * 100, 2
            )
        return self.attendance_percentage

    @property
    def is_below_minimum(self):
        return self.attendance_percentage < self.minimum_required_percentage

    @property
    def is_near_threshold(self):
        return self.attendance_percentage < self.warning_threshold_percentage

from rest_framework import serializers
from .models import AttendanceRecord, AttendanceSummary
from apps.timetable.serializers import CourseSerializer, ClassScheduleSerializer


class AttendanceRecordSerializer(serializers.ModelSerializer):
    """Attendance record serializer"""
    class_schedule = ClassScheduleSerializer(read_only=True)
    course_name = serializers.CharField(source='class_schedule.course.name', read_only=True)
    course_code = serializers.CharField(source='class_schedule.course.code', read_only=True)

    class Meta:
        model = AttendanceRecord
        fields = [
            'id', 'student', 'class_schedule', 'course_name', 'course_code',
            'date', 'status', 'check_in_time', 'check_out_time',
            'minutes_late', 'notes', 'created_at', 'updated_at'
        ]
        read_only_fields = ['student']


class AttendanceSummarySerializer(serializers.ModelSerializer):
    """Attendance summary serializer"""
    course = CourseSerializer(read_only=True)
    status = serializers.SerializerMethodField()
    sessions_needed_for_minimum = serializers.SerializerMethodField()

    class Meta:
        model = AttendanceSummary
        fields = [
            'id', 'student', 'course', 'total_sessions',
            'present_count', 'absent_count', 'late_count', 'excused_count',
            'attendance_percentage', 'minimum_required_percentage',
            'warning_threshold_percentage', 'status',
            'sessions_needed_for_minimum', 'warning_count',
            'last_warning_sent', 'updated_at'
        ]

    def get_status(self, obj):
        if obj.is_below_minimum:
            return 'critical'
        elif obj.is_near_threshold:
            return 'warning'
        return 'good'

    def get_sessions_needed_for_minimum(self, obj):
        """Calculate how many consecutive sessions needed to reach minimum"""
        if obj.attendance_percentage >= obj.minimum_required_percentage:
            return 0

        # Formula: solve for x in (present + x) / (total + x) >= minimum
        present = obj.present_count + obj.late_count
        total = obj.total_sessions - obj.excused_count
        minimum = obj.minimum_required_percentage / 100

        if total == 0:
            return 1

        needed = (minimum * total - present) / (1 - minimum)
        return max(1, int(needed) + 1)

@api_view(['POST'])
@permission_classes([IsAuthenticated])
def mark_attendance(request):
    """
    Mark attendance for a class session
    {
        "class_schedule_id": 1,
        "date": "2026-06-01",
        "status": "present",
        "check_in_time": "08:30",
        "notes": ""
    }
    """
    student = request.user
    data = request.data

    class_schedule = get_object_or_404(ClassSchedule, id=data.get('class_schedule_id'))

    # Check if student is enrolled in this course
    if not StudentCourse.objects.filter(student=student, course=class_schedule.course).exists():
        return Response(
            {'error': 'You are not enrolled in this course'},
            status=status.HTTP_403_FORBIDDEN
        )

    # Create or update attendance record
    record, created = AttendanceRecord.objects.update_or_create(
        student=student,
        class_schedule=class_schedule,
        date=data.get('date'),
        defaults={
            'status': data.get('status', 'present'),
            'check_in_time': data.get('check_in_time'),
            'check_out_time': data.get('check_out_time'),
            'minutes_late': data.get('minutes_late', 0),
            'notes': data.get('notes', '')
        }
    )

    # Update attendance summary
    update_attendance_summary(student, class_schedule.course)

    return Response({
        'message': 'Attendance marked successfully',
        'record': AttendanceRecordSerializer(record).data
    })


def update_attendance_summary(student, course):
    """Update or create attendance summary for a student-course pair"""
    records = AttendanceRecord.objects.filter(student=student, class_schedule__course=course)

    summary, created = AttendanceSummary.objects.get_or_create(
        student=student,
        course=course,
        defaults={
            'minimum_required_percentage': 80.00,
            'warning_threshold_percentage': 85.00
        }
    )

    summary.total_sessions = records.count()
    summary.present_count = records.filter(status='present').count()
    summary.absent_count = records.filter(status='absent').count()
    summary.late_count = records.filter(status='late').count()
    summary.excused_count = records.filter(status='excused').count()

    summary.calculate_percentage()
    summary.save()

    return summary


@api_view(['GET'])
@permission_classes([IsAuthenticated])
def attendance_alerts(request):
    """Get attendance alerts for current student"""
    student = request.user

    summaries = AttendanceSummary.objects.filter(student=student)
    alerts = []

    for summary in summaries:
        if summary.is_below_minimum:
            alerts.append({
                'type': 'critical',
                'course': summary.course.code,
                'course_name': summary.course.name,
                'message': f"Attendance below minimum ({summary.attendance_percentage}% < {summary.minimum_required_percentage}%)",
                'attendance_percentage': summary.attendance_percentage,
                'minimum_required': summary.minimum_required_percentage,
                'sessions_needed': summary.sessions_needed_for_minimum
            })
        elif summary.is_near_threshold:
            alerts.append({
                'type': 'warning',
                'course': summary.course.code,
                'course_name': summary.course.name,
                'message': f"Attendance near threshold ({summary.attendance_percentage}%)",
                'attendance_percentage': summary.attendance_percentage,
                'warning_threshold': summary.warning_threshold_percentage,
                'sessions_needed': summary.sessions_needed_for_minimum
            })

    return Response({
        'total_alerts': len(alerts),
        'critical_count': len([a for a in alerts if a['type'] == 'critical']),
        'warning_count': len([a for a in alerts if a['type'] == 'warning']),
        'alerts': alerts
    })


@api_view(['GET'])
@permission_classes([IsAuthenticated])
def attendance_report(request, course_id):
    """Get detailed attendance report for a specific course"""
    student = request.user

    records = AttendanceRecord.objects.filter(
        student=student,
        class_schedule__course_id=course_id
    ).order_by('-date')

    summary = get_object_or_404(
        AttendanceSummary,
        student=student,
        course_id=course_id
    )

    # Weekly breakdown
    from collections import defaultdict
    weekly_data = defaultdict(lambda: {'present': 0, 'absent': 0, 'late': 0, 'total': 0})

    for record in records:
        week_key = record.date.strftime('%Y-W%U')
        weekly_data[week_key]['total'] += 1
        weekly_data[week_key][record.status] += 1

    return Response({
        'course': summary.course.code,
        'course_name': summary.course.name,
        'summary': AttendanceSummarySerializer(summary).data,
        'recent_records': AttendanceRecordSerializer(records[:10], many=True).data,
        'weekly_breakdown': dict(weekly_data)
    })

@api_view(['POST'])
@permission_classes([IsAuthenticated])
def generate_study_plan(request):
    """
    Generate an optimized study plan using Google OR-Tools CP-SAT solver

    Request body:
    {
        "week_start": "2026-06-01",
        "week_end": "2026-06-07",
        "courses": [1, 2, 3],  // optional, defaults to enrolled courses
        "preferences": {
            "max_hours_per_day": 6,
            "preferred_times": ["morning", "afternoon"],
            "min_break_minutes": 30
        }
    }
    """
    import time as time_module
    start_time = time_module.time()

    student = request.user
    data = request.data

    week_start = datetime.strptime(data.get('week_start'), '%Y-%m-%d').date()
    week_end = datetime.strptime(data.get('week_end'), '%Y-%m-%d').date()

    # Get student's courses
    if 'courses' in data:
        course_ids = data['courses']
    else:
        course_ids = StudentCourse.objects.filter(
            student=student,
            semester='semester_2',
            academic_year='2026'
        ).values_list('course_id', flat=True)

    courses = Course.objects.filter(id__in=course_ids)

    if not courses.exists():
        return Response(
            {'error': 'No courses found. Please enroll in courses first.'},
            status=status.HTTP_400_BAD_REQUEST
        )

    # Get preferences
    prefs = data.get('preferences', {})
    max_hours_per_day = prefs.get('max_hours_per_day', student.max_study_hours_per_day or 6)
    min_break = prefs.get('min_break_minutes', student.preferred_break_duration or 30)
    study_preference = student.study_preference or 'flexible'

    # Get existing class schedules for the week
    existing_schedules = ClassSchedule.objects.filter(
        course_id__in=course_ids,
        is_recurring=True
    )

    # Get upcoming assignments
    assignments = Assignment.objects.filter(
        course_id__in=course_ids,
        due_date__gte=datetime.combine(week_start, time.min),
        due_date__lte=datetime.combine(week_end, time.max)
    )

    # ============ OR-TOOLS CONSTRAINT SOLVER ============
    model = cp_model.CpModel()

    # Define time slots: 30-minute blocks from 6:00 to 23:00
    SLOT_DURATION = 30  # minutes
    START_HOUR = 6
    END_HOUR = 23
    SLOTS_PER_DAY = (END_HOUR - START_HOUR) * 2  # 34 slots per day
    DAYS = 7  # Monday to Sunday

    # Create decision variables: x[course][day][slot] = 0 or 1
    study_slots = {}
    for course in courses:
        for day in range(DAYS):
            for slot in range(SLOTS_PER_DAY):
                study_slots[(course.id, day, slot)] = model.NewBoolVar(
                    f'study_c{course.id}_d{day}_s{slot}'
                )

    # ============ HARD CONSTRAINTS ============

    # 1. No overlap with existing classes
    for schedule in existing_schedules:
        day = schedule.day_of_week
        start_slot = (schedule.start_time.hour - START_HOUR) * 2 + schedule.start_time.minute // 30
        end_slot = (schedule.end_time.hour - START_HOUR) * 2 + schedule.end_time.minute // 30

        for course in courses:
            for slot in range(max(0, start_slot), min(SLOTS_PER_DAY, end_slot)):
                model.Add(study_slots[(course.id, day, slot)] == 0)

    # 2. Maximum hours per day
    for day in range(DAYS):
        daily_slots = []
        for course in courses:
            for slot in range(SLOTS_PER_DAY):
                daily_slots.append(study_slots[(course.id, day, slot)])
        model.Add(sum(daily_slots) <= max_hours_per_day * 2)  # *2 because slots are 30 min

    # 3. Minimum break between study sessions (30 min = 1 slot)
    for course in courses:
        for day in range(DAYS):
            for slot in range(SLOTS_PER_DAY - 1):
                # If studying at slot, can't study at slot+1 (break requirement)
                # Relaxed: allow consecutive if same course
                pass  # Simplified for this version

    # 4. Each course gets minimum study time based on credits and difficulty
    for course in courses:
        min_slots = course.credits * 2  # 1 hour per credit per week minimum
        course_slots = []
        for day in range(DAYS):
            for slot in range(SLOTS_PER_DAY):
                course_slots.append(study_slots[(course.id, day, slot)])
        model.Add(sum(course_slots) >= min_slots)

    # ============ SOFT CONSTRAINTS / OBJECTIVE ============

    # Objective: Maximize study quality score
    # Score = difficulty_weight * preference_alignment * deadline_proximity
    objective_terms = []

    for course in courses:
        difficulty = course.difficulty_rating or 3

        for day in range(DAYS):
            for slot in range(SLOTS_PER_DAY):
                # Time preference score
                hour = START_HOUR + slot // 2
                preference_score = get_preference_score(hour, study_preference)

                # Deadline proximity score (higher if assignment due soon)
                deadline_score = get_deadline_score(course, assignments, day)

                # Combined score
                score = int(difficulty * preference_score * deadline_score * 10)
                objective_terms.append(score * study_slots[(course.id, day, slot)])

    model.Maximize(sum(objective_terms))

    # ============ SOLVE ============
    solver = cp_model.CpSolver()
    solver.parameters.max_time_in_seconds = 5.0  # Max 5 seconds
    solver.parameters.num_search_workers = 4

    result = solver.Solve(model)

    if result not in [cp_model.OPTIMAL, cp_model.FEASIBLE]:
        return Response(
            {'error': 'Could not generate a feasible study plan. Try relaxing constraints.'},
            status=status.HTTP_400_BAD_REQUEST
        )

    # ============ CREATE STUDY PLAN ============
    study_plan = StudyPlan.objects.create(
        student=student,
        name=f"Study Plan: {week_start} to {week_end}",
        week_start=week_start,
        week_end=week_end,
        status='active',

        optimization_score=round(solver.ObjectiveValue() / 100, 2) if result == cp_model.OPTIMAL else None

        if not StudentCourse.objects.filter(student=student, course=group.course).exists():
            return Response(
                {'error': 'You must be enrolled in this course to join the study group'},
                status=status.HTTP_403_FORBIDDEN
            )

        # Determine membership status based on privacy
        membership_status = 'active' if group.privacy == 'public' else 'pending'

        membership = GroupMembership.objects.create(
            group=group,
            student=student,
            status=membership_status,
            role='member'
        )

        # Update group status if now full
        if group.is_full:
            group.status = 'full'
            group.save()

        return Response({
            'message': 'Join request submitted' if membership_status == 'pending' else 'Successfully joined group',
            'membership': GroupMembershipSerializer(membership).data
        })


class LeaveGroupView(generics.DestroyAPIView):
    """Leave a study group"""
    permission_classes = [IsAuthenticated]

    def delete(self, request, group_id):
        group = get_object_or_404(StudyGroup, id=group_id)
        membership = get_object_or_404(
            GroupMembership,
            group=group,
            student=request.user,
            status='active'
        )

        membership.status = 'left'
        membership.left_at = datetime.now()
        membership.save()

        # Update group status if was full
        if group.status == 'full':
            group.status = 'active'
            group.save()

        return Response({'message': 'Successfully left the group'})

def find_study_groups(request):
    """
    Find recommended study groups for current student
    Uses compatibility scoring based on:
    - Shared courses
    - Temporal overlap in free time
    - Group size preference
    """
    student = request.user

    # Get student's enrolled courses
    enrolled_courses = StudentCourse.objects.filter(student=student).values_list('course_id', flat=True)

    # Get student's class schedules (busy times)
    busy_slots = get_student_busy_slots(student)

    # Find active groups for enrolled courses
    groups = StudyGroup.objects.filter(
        course_id__in=enrolled_courses,
        status='active',
        privacy__in=['public', 'invite_only']
    ).exclude(
        members__student=student  # Exclude groups already joined
    )

    # Score each group
    scored_groups = []
    for group in groups:
        score = calculate_group_compatibility(student, group, busy_slots)
        scored_groups.append({
            'group': StudyGroupSerializer(group).data,
            'compatibility_score': score,
            'reasons': get_compatibility_reasons(student, group, busy_slots)
        })

    # Sort by compatibility score (highest first)
    scored_groups.sort(key=lambda x: x['compatibility_score'], reverse=True)

    return Response({
        'recommended_groups': scored_groups[:10],
        'total_available': len(scored_groups)
    })


def get_student_busy_slots(student):
    """Get student's busy time slots for the week"""
    from collections import defaultdict

    busy = defaultdict(list)
    schedules = ClassSchedule.objects.filter(
        course__enrolled_students__student=student
    )

    for schedule in schedules:
        busy[schedule.day_of_week].append({
            'start': schedule.start_time,
            'end': schedule.end_time
        })

    return busy

        (0, 'Monday'),

@api_view(['GET'])
@permission_classes([IsAuthenticated])
def detect_conflicts(request):
    """
    Detect scheduling conflicts for the current student
    Returns: overlapping classes, back-to-back exams, deadline clustering
    """
    student = request.user

    # Get student's class schedules
    student_courses = StudentCourse.objects.filter(student=student).values_list('course_id', flat=True)
    schedules = ClassSchedule.objects.filter(course_id__in=student_courses)
    exams = ExamSchedule.objects.filter(course_id__in=student_courses)
    assignments = Assignment.objects.filter(course_id__in=student_courses, due_date__gte=datetime.now())

    conflicts = {
        'overlapping_classes': [],
        'back_to_back_exams': [],
        'deadline_clustering': [],
        'total_conflicts': 0
    }

    # Detect overlapping classes
    schedule_list = list(schedules)
    for i in range(len(schedule_list)):
        for j in range(i + 1, len(schedule_list)):
            s1, s2 = schedule_list[i], schedule_list[j]
            if (s1.day_of_week == s2.day_of_week and
                s1.start_time < s2.end_time and
                s2.start_time < s1.end_time):
                conflicts['overlapping_classes'].append({
                    'class_1': ClassScheduleSerializer(s1).data,
                    'class_2': ClassScheduleSerializer(s2).data,
                    'conflict_type': 'overlap',
                    'severity': 'critical'
                })

    # Detect back-to-back exams (< 2 hours gap)
    exam_list = list(exams)
    for i in range(len(exam_list)):
        for j in range(i + 1, len(exam_list)):
            e1, e2 = exam_list[i], exam_list[j]
            if e1.date == e2.date:
                from datetime import datetime, timedelta
                t1_end = datetime.combine(e1.date, e1.end_time)
                t2_start = datetime.combine(e2.date, e2.start_time)
                gap = (t2_start - t1_end).total_seconds() / 3600

                if 0 <= gap < 2:
                    conflicts['back_to_back_exams'].append({
                        'exam_1': ExamScheduleSerializer(e1).data,
                        'exam_2': ExamScheduleSerializer(e2).data,
                        'gap_hours': round(gap, 2),
                        'conflict_type': 'back_to_back',
                        'severity': 'warning'
                    })

    # Detect deadline clustering (3+ assignments within 3 days)
    assignment_list = list(assignments.order_by('due_date'))
    for i in range(len(assignment_list)):
        cluster = [assignment_list[i]]
        for j in range(i + 1, len(assignment_list)):
            diff = (assignment_list[j].due_date - assignment_list[i].due_date).days
            if diff <= 3:
                cluster.append(assignment_list[j])

        if len(cluster) >= 3:
            conflicts['deadline_clustering'].append({
                'assignments': AssignmentSerializer(cluster, many=True).data,
                'cluster_size': len(cluster),
                'date_range': f"{cluster[0].due_date.date()} to {cluster[-1].due_date.date()}",
                'conflict_type': 'deadline_cluster',
                'severity': 'warning'
            })

    conflicts['total_conflicts'] = (
        len(conflicts['overlapping_classes']) +
        len(conflicts['back_to_back_exams']) +
        len(conflicts['deadline_clustering'])
    )

    return Response(conflicts)
